public void decToBinary(int num){

	int[] arr = new int[10];
	int cnt = 0;
	while(num > 0){
		int rem = num % 2;
		arr[cnt] = rem;
		num /= 2;
		cnt++;
	}
	
	for(int i = cnt-1; i >=0; i--){
		System.out.print(arr[i]);
	}
	
}

import java.util.*;
class Q4{

public int BinToDec(String s){
	int num = 0;
	int Base = 1;
	int i = s.length()-1;
	while(i >= 0){
		if(s.charAt(i) == '1'){
			num += Base;
		}
		Base *= 2;
		i--;
	}
	
	return num;
	
}


public static void main(String[] args){
	Scanner sc = new Scanner(System.in); 

	Q4 obj = new Q4();
	System.out.println("Enter Binary Number : ");
	String num = sc.nextLine();
	System.out.println("Binary to Decimal : " + obj.BinToDec(num));

import java.util.*;
class Q9{
public static void main(String args[]){
	int n = 1000;
	int[] arr = new int[n];
	for(int i =0; i < n; i++){
		arr[i] = 0;
	}

	for(int i = 2; i < n; i++){
		for(int j = 2; j < i; j++){
			if(i % j == 0){
				arr[i] = 1;
				break;
			}
			
		}
	}
	
	int sum = 0;
	int cnt = 0;
	for(int i = 2; i < n; i++){
		if(arr[i] == 0){
			System.out.print(i + " ");
			sum += i;
			cnt++;
		}
		if(cnt >= 100)break;
	}
	System.out.println("\n sum of first 100 prime number is " + sum);
}
}

import java.util.*;
class palindrome{
public static void main(String args[]){
	Scanner sc = new Scanner(System.in);
	System.out.print("Enter number : ");
	int num = sc.nextInt();
	
	int n = num;
	int reverse = 0;
	
	while(n > 0){
		int digit = n % 10;
		reverse *= 10;
		reverse += digit;
		n /= 10;
	}

	if(reverse == num){
		System.out.println("Number is Palindrome ");
	}
	else{
		System.out.println("Number is Not Palindrome ");
	}
}
}
