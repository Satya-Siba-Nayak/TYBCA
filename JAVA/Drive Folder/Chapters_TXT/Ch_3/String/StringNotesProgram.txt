/* Example Demonstrte getChar() method of String Class */

import java.util.*;
import java.io.*;
import java.lang.String;
public class StringNotesProgram
{
  public static void main(String[] args)throws IOException
	{
  		String str1 = "Hi Abhjit how are you";
		String str2 = "Hi abhjit how are you";
		char ch= str1.charAt(5);
		System.out.println("The retrived character is:   "+ch);
  		char[] arr = new char[7];
  		//Syntax is ** getChar(int sourcebegin int sourceend,char[] destination , int destinationbegin)
		str1.getChars(3, 9, arr, 0);
  		System.out.print("The retrived String is:     ");
		System.out.println(arr);
  		

		//Check String equality
		boolean check= str1.equals(str2);
		System.out.println("Is two Strings found Equal  "+check);

		// Check String equality by ibnoring case(Upper case and lower case)
		boolean check1= str1.equalsIgnoreCase(str2);
		System.out.println("Is two Strings found Equal  "+check1);
		
		// Check String using CompareTO()
		
		int comare= str1.compareTo(str2);
		System.out.println("Output after applying compareTo function    "+comare);
		
		// Check String using compareToIgnoreCase()
		
		int comare1= str1.compareToIgnoreCase(str2);
		System.out.println("Output after applying compareToIgnoreCase function    "+comare1);

		// String concat
		String str3= str1.concat(str2);
		System.out.println("String concatinate is  "+str3);
 	 }
}
