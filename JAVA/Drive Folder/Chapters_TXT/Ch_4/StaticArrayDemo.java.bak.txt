// Programm illustrate  a use of static initialize block

import java.io.*;
class Block
{
	static int [] x ;
	static 
	{
		x = new int [5];
		for (int i=0 ; i<5 ; i++ )
			x[i] = i
	}
} 
class StaticArrayDemo
{
	public static void main(String[] args) 
	{
		for (int i=0 ; i<5 ;i++ )
			System.out.println (Block.x[i]);

	}
}
