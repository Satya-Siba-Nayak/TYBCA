import java.io.*;
class Q 
{
	void display()
	{
		System.out.println("Good Morning");
	}
} 
class R extends Q
{
}
class S extends R
{
	void display()
	{
		System.out.println("Good Afternoon");
	}
}
class MethodOverriding
{
	public static void main(String[] args) 
	{
		S s1 = new S();
		s1.display();
	}
}
